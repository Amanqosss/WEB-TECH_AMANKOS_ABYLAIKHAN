-- Create a domain-specific schema
CREATE SCHEMA IF NOT EXISTS car_rental;

SET search_path TO car_rental;


-- ============================================================
-- TABLE DEFINITIONS (CREATE TABLE IF NOT EXISTS)
-- ============================================================

-- Table: brands
-- Stores car manufacturer brands (e.g. Toyota, BMW)
-- VARCHAR(50) chosen — brand names rarely exceed 50 chars
CREATE TABLE IF NOT EXISTS car_rental.brands (
    brand_id    SERIAL PRIMARY KEY,
    brand_name  VARCHAR(20) NOT NULL,                           -- NOT NULL: every brand must have a name
    CONSTRAINT uq_brand_name UNIQUE (brand_name)               -- UNIQUE: no duplicate brand names
);

-- Table: vehicle_types
-- Stores vehicle categories: sedan, SUV, truck, etc.
CREATE TABLE IF NOT EXISTS car_rental.vehicle_types (
    type_id     SERIAL PRIMARY KEY,
    type_name   VARCHAR(50) NOT NULL,
    CONSTRAINT uq_type_name UNIQUE (type_name)
);

-- Table: rental_locations
-- Physical locations where vehicles are rented/returned
CREATE TABLE IF NOT EXISTS car_rental.rental_locations (
    location_id     SERIAL PRIMARY KEY,
    location_name   VARCHAR(50) NOT NULL,
    address         TEXT
);

-- Table: models
-- Car models tied to a brand (e.g. Toyota Camry)
-- VARCHAR(100) for model_name — some full model names are long
CREATE TABLE IF NOT EXISTS car_rental.models (
    model_id    SERIAL PRIMARY KEY,
    brand_id    INT NOT NULL,
    model_name  VARCHAR(100) NOT NULL,
    CONSTRAINT fk_model_brand FOREIGN KEY (brand_id) REFERENCES car_rental.brands(brand_id)
);

-- Table: vehicles
-- Individual vehicles available for rental
-- status DEFAULT 'available' — sensible initial state for a new vehicle
CREATE TABLE IF NOT EXISTS car_rental.vehicles (
    vehicle_id      SERIAL PRIMARY KEY,
    type_id         INT NOT NULL,
    location_id     INT NOT NULL,
    model_id        INT NOT NULL,
    year            INT NOT NULL,
    license_plate   VARCHAR(20) NOT NULL,
    daily_rate      NUMERIC(10,2) NOT NULL,                    -- price per day
    status          VARCHAR(20) NOT NULL DEFAULT 'available',  -- DEFAULT value
    CONSTRAINT fk_vehicle_type      FOREIGN KEY (type_id)     REFERENCES car_rental.vehicle_types(type_id),
    CONSTRAINT fk_vehicle_location  FOREIGN KEY (location_id) REFERENCES car_rental.rental_locations(location_id),
    CONSTRAINT fk_vehicle_model     FOREIGN KEY (model_id)    REFERENCES car_rental.models(model_id),
    CONSTRAINT chk_vehicle_status   CHECK (status IN ('available', 'rented', 'maintenance')), -- specific options
    CONSTRAINT chk_daily_rate       CHECK (daily_rate >= 0),   -- measured value cannot be negative
    CONSTRAINT uq_license_plate     UNIQUE (license_plate)     -- UNIQUE: no two vehicles share a plate
);

-- Table: customers
-- People who rent vehicles
CREATE TABLE IF NOT EXISTS car_rental.customers (
    customer_id     SERIAL PRIMARY KEY,
    surname         VARCHAR(20) NOT NULL,
    name            VARCHAR(20) NOT NULL,                      -- NOT NULL constraint
    email           VARCHAR(100) NOT NULL,
    phone           VARCHAR(20),
    driver_license  VARCHAR(50) NOT NULL,
    CONSTRAINT uq_customer_email    UNIQUE (email),
    CONSTRAINT uq_driver_license    UNIQUE (driver_license)
);

-- Table: employees
-- Staff working at rental locations
CREATE TABLE IF NOT EXISTS car_rental.employees (
    employee_id     SERIAL PRIMARY KEY,
    location_id     INT,
    surname         VARCHAR(20) NOT NULL,
    name            VARCHAR(20) NOT NULL,
    position        VARCHAR(20),
    salary          NUMERIC(10,2) NOT NULL DEFAULT 0,
    CONSTRAINT fk_employee_location FOREIGN KEY (location_id) REFERENCES car_rental.rental_locations(location_id),
    CONSTRAINT chk_employee_salary  CHECK (salary >= 0)        -- salary cannot be negative
);

-- Table: reservations
-- A customer reserves a vehicle before formal rental
-- reservation_date DEFAULT CURRENT_DATE — auto-set to today
CREATE TABLE IF NOT EXISTS car_rental.reservations (
    reservation_id      SERIAL PRIMARY KEY,
    customer_id         INT NOT NULL,
    vehicle_id          INT NOT NULL,
    reservation_date    DATE NOT NULL DEFAULT CURRENT_DATE,    -- DEFAULT value
    status              VARCHAR(50) DEFAULT 'pending',
    CONSTRAINT fk_res_customer  FOREIGN KEY (customer_id) REFERENCES car_rental.customers(customer_id),
    CONSTRAINT fk_res_vehicle   FOREIGN KEY (vehicle_id)  REFERENCES car_rental.vehicles(vehicle_id),
    CONSTRAINT chk_res_date     CHECK (reservation_date >= '2026-01-01') -- date must be after 2026-01-01
);

-- Table: rentals
-- Formal rental agreement: customer, vehicle, employee, dates
CREATE TABLE IF NOT EXISTS car_rental.rentals (
    rental_id       SERIAL PRIMARY KEY,
    customer_id     INT NOT NULL,
    vehicle_id      INT NOT NULL,
    employee_id     INT,
    start_date      DATE NOT NULL,
    end_date        DATE,
    CONSTRAINT fk_rental_customer   FOREIGN KEY (customer_id)  REFERENCES car_rental.customers(customer_id),
    CONSTRAINT fk_rental_vehicle    FOREIGN KEY (vehicle_id)   REFERENCES car_rental.vehicles(vehicle_id),
    CONSTRAINT fk_rental_employee   FOREIGN KEY (employee_id)  REFERENCES car_rental.employees(employee_id),
    CONSTRAINT chk_rental_dates     CHECK (end_date IS NULL OR end_date >= start_date)
);

-- Table: payments
-- Payments linked to a rental
-- tax GENERATED ALWAYS AS 12% of amount — computed automatically, never stored manually
CREATE TABLE IF NOT EXISTS car_rental.payments (
    payment_id      SERIAL PRIMARY KEY,
    rental_id       INT NOT NULL,
    amount          NUMERIC(10,2) NOT NULL,
    tax             NUMERIC(10,2) GENERATED ALWAYS AS (amount * 0.12) STORED, -- GENERATED column: 12% VAT
    payment_date    DATE NOT NULL DEFAULT CURRENT_DATE,        -- DEFAULT value
    method          VARCHAR(10),
    CONSTRAINT fk_payment_rental    FOREIGN KEY (rental_id)    REFERENCES car_rental.rentals(rental_id),
    CONSTRAINT chk_amount_positive  CHECK (amount > 0),        -- amount cannot be negative
    CONSTRAINT chk_payment_method   CHECK (method IN ('cash', 'card', 'online'))
);

-- Table: vehicle_conditions
-- Condition checks on a vehicle (before/after rental)
CREATE TABLE IF NOT EXISTS car_rental.vehicle_conditions (
    condition_id        SERIAL PRIMARY KEY,
    vehicle_id          INT NOT NULL,
    check_date          DATE NOT NULL,
    condition_status    VARCHAR(50),
    notes               TEXT,
    CONSTRAINT fk_cond_vehicle  FOREIGN KEY (vehicle_id) REFERENCES car_rental.vehicles(vehicle_id),
    CONSTRAINT chk_cond_date    CHECK (check_date >= '2026-01-01') -- date after 2026-01-01
);

-- Table: maintenance_records
-- Maintenance history for each vehicle
CREATE TABLE IF NOT EXISTS car_rental.maintenance_records (
    maintenance_id      SERIAL PRIMARY KEY,
    vehicle_id          INT NOT NULL,
    check_date          DATE NOT NULL,
    condition_status    VARCHAR(50),
    notes               TEXT,
    cost                NUMERIC(10,2) NOT NULL DEFAULT 0,
    CONSTRAINT fk_maint_vehicle FOREIGN KEY (vehicle_id) REFERENCES car_rental.vehicles(vehicle_id),
    CONSTRAINT chk_maint_cost   CHECK (cost >= 0)              -- maintenance cost cannot be negative
);


-- ============================================================
-- ALTER TABLE OPERATIONS (7 operations with explanatory comments)
-- ============================================================

-- 1. brand_name was VARCHAR(20) — too short for names like 'Mercedes-Benz'
ALTER TABLE car_rental.brands
    ALTER COLUMN brand_name TYPE VARCHAR(50);

-- 2. phone in customers was VARCHAR(20) — international formats like +7 (701) 123-45-67 need more space
ALTER TABLE car_rental.customers
    ALTER COLUMN phone TYPE VARCHAR(30);

-- 3. position in employees was VARCHAR(20) — too short for 'Senior Fleet Manager'
ALTER TABLE car_rental.employees
    ALTER COLUMN position TYPE VARCHAR(60);

-- 4. Adding 'notes' column to reservations — needed to record special customer requests
ALTER TABLE car_rental.reservations
    ADD COLUMN IF NOT EXISTS notes TEXT;

-- 5. Adding unique constraint on reservations to prevent duplicate bookings
--    (same customer + same vehicle on the same date)
ALTER TABLE car_rental.reservations
    ADD CONSTRAINT uq_reservation UNIQUE (customer_id, vehicle_id, reservation_date);

-- 6. Renaming 'name' to 'first_name' in customers for clarity and consistency with industry standards
ALTER TABLE car_rental.customers
    RENAME COLUMN name TO first_name;

-- 7. Renaming 'name' to 'first_name' in employees for the same reason
ALTER TABLE car_rental.employees
    RENAME COLUMN name TO first_name;
